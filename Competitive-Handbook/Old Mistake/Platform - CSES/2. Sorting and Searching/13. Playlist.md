#include <bits/stdc++.h>

#ifdef LOCAL
#include "schematics.hpp"
static_assert(schematics::invariant);
#else
#define debug(...)
#endif

using namespace std;

void solve() {
    int n;
    cin >> n;
    unordered_map<int, int> dasmap;
    dasmap.reserve(2 * 1e5);
    vector<int> data(n, 0);

    for (int i = 0; i < n; i++) {
        cin >> data[i];
    }

    if (n == 1) {
        cout << 1;
        return;
    } else if (n == 2) {
        if (data.front() != data.back()) {
            cout << 2;
        } else {
            cout << 1;
        }
        return;
    }

    int maks = 0;
    int l = 0, r = 0;

    while (r < n) {
        if (!dasmap[data[r]]) {
            dasmap[data[r]] = 1;
            r++;
        } else {
            dasmap[data[l]] = 0;
            l++;
        }

        maks = max(maks, r - l);
    }

    cout << maks;
}

auto main() -> int {
    std::ios_base::sync_with_stdio(false);
    std::cin.tie(nullptr);
    int t = 1;
    // cin >> t;
    while (t--) {
        solve();
    }
    return 0;
}
