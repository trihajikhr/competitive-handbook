#include <algorithm>
#include <iostream>
#include <utility>
#include <vector>

#ifdef LOCAL
#include "schematics.hpp"
static_assert(schematics::invariant);
#else
#define debug(...)
#endif

using namespace std;

void solve() {
    int n;
    cin >> n;
    vector<pair<int, int>> event(n);
    for (int i = 0; i < n; i++) {
        cin >> event[i].first >> event[i].second;
    }
    sort(event.begin(), event.end(),
         [](const pair<int, int>& a, const pair<int, int>& b) { return a.second < b.second; });

    int ans = 0, curr = 0;
    ans++, curr = event.front().second;
    for (int i = 1; i < n; i++) {
        if (event[i].first >= curr) {
            ans++;
            curr = event[i].second;
        }
    }

    cout << ans;
}

auto main() -> int {
    std::ios_base::sync_with_stdio(false);
    std::cin.tie(nullptr);
    int t = 1;
    // cin >> t;
    while (t--) {
        solve();
    }
    return 0;
}
