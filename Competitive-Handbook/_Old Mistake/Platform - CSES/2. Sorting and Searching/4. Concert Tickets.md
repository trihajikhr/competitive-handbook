# Concert Tickets

## Problem

## Solution

```cpp
#include <algorithm>
#include <iostream>
#include <set>
#include <vector>
using namespace std;

auto main() -> int {
    std::ios_base::sync_with_stdio(false);
    std::cin.tie(nullptr);
    int n, m;
    cin >> n >> m;
    vector<int> tick_temp(n), per(m);

    for (int i = 0; i < n; i++) {
        cin >> tick_temp[i];
    }

    for (int i = 0; i < m; i++) {
        cin >> per[i];
    }

    sort(tick_temp.begin(), tick_temp.end());
    multiset<int> tick(tick_temp.begin(), tick_temp.end());

    for (int i = 0; i < m; i++) {
        auto it = tick.upper_bound(per[i]);
        if (it == tick.begin()) {
            cout << -1 << "\n";
        } else {
            --it;
            cout << *it << "\n";
            tick.erase(it);
        }
    }
    return 0;
}
```
