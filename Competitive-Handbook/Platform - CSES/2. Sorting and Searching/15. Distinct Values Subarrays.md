#include <bits/stdc++.h>

#ifdef LOCAL
#include "schematics.hpp"
static_assert(schematics::invariant);
#else
#define debug(...)
#endif

using namespace std;

void solve() {
    int n;
    cin >> n;
    vector<int> data(n);
    for (int i = 0; i < n; i++) {
        cin >> data[i];
    }

    set<int> daset;
    long long ans = 0;
    int l = 0, r = 0;
    while (r < n) {
        if (!daset.count(data[r])) {
            daset.insert(data[r]);
            r++;
            ans += r - l;
        } else {
            while (daset.count(data[r])) {
                daset.erase(data[l]);
                l++;
            }
            daset.insert(data[r]);
            r++;
            ans += r - l;
        }
    }
    cout << ans;
}

auto main() -> int {
    std::ios_base::sync_with_stdio(false);
    std::cin.tie(nullptr);
    int t = 1;
    // cin >> t;
    while (t--) {
        solve();
    }
    return 0;
}
