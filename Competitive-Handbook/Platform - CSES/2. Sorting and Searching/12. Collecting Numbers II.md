#include <bits/stdc++.h>

#ifdef LOCAL
#include "schematics.hpp"
static_assert(schematics::invariant);
#else
#define debug(...)
#endif

using namespace std;

void solve() {
    int n, m;
    cin >> n >> m;
    vector<int> real(n + 1);
    vector<pair<int, int>> data(n + 1);

    for (int i = 1; i <= n; i++) {
        cin >> real[i];
        data[i] = {real[i], i};
    }

    sort(data.begin(), data.end());
    int cnt = 1;
    for (int i = 1; i <= n - 1; i++) {
        cnt += data[i].second > data[i + 1].second;
    }

    while (m--) {
        int a, b;
        cin >> a >> b;

        set<int> daset = {real[a], real[b]};
        if (real[a] + 1 <= n) {
            daset.insert(real[a] + 1);
        }

        if (real[b] + 1 <= n) {
            daset.insert(real[b] + 1);
        }

        if (real[a] - 1 >= 1) {
            daset.insert(real[a] - 1);
        }

        if (real[b] - 1 >= 1) {
            daset.insert(real[b] - 1);
        }

        for (const auto& val : daset) {
            if (val < n && data[val].second > data[val + 1].second) {
                cnt--;
            }
        }

        swap(data[real[a]].second, data[real[b]].second);
        swap(real[a], real[b]);

        for (const auto& val : daset) {
            if (val < n && data[val].second > data[val + 1].second) {
                cnt++;
            }
        }

        cout << cnt << "\n";
    }
}

auto main() -> int {
    std::ios_base::sync_with_stdio(false);
    std::cin.tie(nullptr);
    int t = 1;
    // cin >> t;
    while (t--) {
        solve();
    }
    return 0;
}
