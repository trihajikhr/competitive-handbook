# Maximum Subarray Sum

## Problem


## Solution

```cpp
#include <algorithm>
#include <iostream>
#include <vector>

#ifdef LOCAL
#include "schematics.hpp"
static_assert(schematics::invariant);
#else
#define debug(...)
#endif

using namespace std;

void solve() {
    int n;
    cin >> n;
    vector<long long> vec(n);

    for (int i = 0; i < n; i++) {
        cin >> vec[i];
    }

    long long curr_maks = vec.front();
    long long global_maks = vec.front();

    for (int i = 1; i < n; i++) {
        curr_maks = max(curr_maks + vec[i], vec[i]);
        global_maks = max(global_maks, curr_maks);
    }

    cout << global_maks;
}

auto main() -> int {
    std::ios_base::sync_with_stdio(false);
    std::cin.tie(nullptr);
    int t = 1;
    // cin >> t;
    while (t--) {
        solve();
    }
    return 0;
}
```
