#include <bits/stdc++.h>

#ifdef LOCAL
#include "schematics.hpp"
static_assert(schematics::invariant);
#else
#define debug(...)
#endif

using namespace std;

void solve() {
    int x, n;
    cin >> x >> n;
    vector<int> lamp(n);
    for (int i = 0; i < n; i++) {
        cin >> lamp[i];
    }

    set<int> pos;
    multiset<int> leng;
    pos.insert(0);
    pos.insert(x);
    leng.insert(x);

    for (int i = 0; i < n; i++) {
        pos.insert(lamp[i]);
        auto it = pos.lower_bound(lamp[i]);
        int hi = *++it;
        advance(it, -2);
        int lo = *it;

        leng.erase(leng.find(hi - lo));
        leng.insert(hi - lamp[i]);
        leng.insert(lamp[i] - lo);
        cout << *--leng.end() << " ";
    }
}

auto main() -> int {
    std::ios_base::sync_with_stdio(false);
    std::cin.tie(nullptr);
    int t = 1;
    // acin >> t;
    while (t--) {
        solve();
    }
    return 0;
}
