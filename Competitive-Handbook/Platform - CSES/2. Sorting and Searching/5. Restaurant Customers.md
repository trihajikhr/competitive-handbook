# Restaurant Customers

## Solution

```cpp
#include <algorithm>
#include <iostream>
#include <vector>
using namespace std;

void solve() {
    int n;
    cin >> n;
    vector<pair<int, int>> event(n * 2);
    for (int i = 0, a, b; i < n; i++) {
        cin >> a >> b;
        event[i * 2] = {a, 1};
        event[i * 2 + 1] = {b, 0};
    }

    sort(event.begin(), event.end());
    int curr = 0, maks = 0;
    for (int i = 0; i < n * 2; i++) {
        curr += event[i].second == 1 ? 1 : -1;
        maks = max(maks, curr);
    }
    cout << maks;
}

auto main() -> int {
    std::ios_base::sync_with_stdio(false);
    std::cin.tie(nullptr);
    int t = 1;
    // cin >> t;
    while (t--) {
        solve();
    }
    return 0;
}
```
