#include <iostream>
#include <map>
#include <vector>

#ifdef LOCAL
#include "schematics.hpp"
static_assert(schematics::invariant);
#else
#define debug(...)
#endif

using namespace std;

void solve() {
    long long n, x;
    cin >> n >> x;
    map<long long, vector<int>> dasmap;

    for (int i = 1; i <= n; i++) {
        long long val;
        cin >> val;
        dasmap[val].push_back(i);
    }

    for (const auto& [val, idx] : dasmap) {
        long long temp = x - val;
        if (val < x && dasmap.count(temp)) {
            if (!dasmap[temp].empty() && (dasmap[temp].front() != idx.front())) {
                cout << dasmap[temp].front() << " " << idx.front();
                return;
            } else if (!dasmap[temp].empty() && (dasmap[temp].front() == idx.front())) {
                if (idx.size() > 1) {
                    cout << idx[0] << " " << idx[1];
                    return;
                }
            }
        }
    }

    cout << "IMPOSSIBLE";
}

auto main() -> int {
    std::ios_base::sync_with_stdio(false);
    std::cin.tie(nullptr);
    int t = 1;
    // cin >> t;
    while (t--) {
        solve();
    }
    return 0;
}
